CREATE TABLE users (
    user_id VARCHAR(255) PRIMARY KEY,
    create_date DATE DEFAULT CURRENT_DATE
);

CREATE TABLE admins (
    admin_id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL,
    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users (user_id)
);

CREATE TABLE communities(
    community_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    created_by VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL UNIQUE,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    member_count INT DEFAULT 1,

    FOREIGN KEY (created_by) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE community_members(
    community_id INT NOT NULL,
    user_id VARCHAR(255) NOT NULL,
    joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    PRIMARY KEY (community_id, user_id),
    
    FOREIGN KEY (community_id) REFERENCES communities (community_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE posts(
    post_id BIGINT GENERATED AS ALWAYS AS IDENTITY PRIMARY KEY,
    community_id INT,
    user_id VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE,
    like_count INT DEFAULT 0,
    comment_count INT DEFAULT 0,

    FOREIGN KEY (community_id) REFERENCES communities (community_id),  
    FOREIGN KEY (user_id) REFERENCES users (user_id)
);

CREATE TABLE comments(
    comment_id BIGINT GENERATED AS ALWAYS AS IDENTITY PRIMARY KEY,
    post_id BIGINT NOT NULL,
    user_id VARCHAR(255) NOT NULL,
    parent_comment_id BIGINT,
    content TEXT NOT NULL,
    created_at TIMESTAMP CURRENT_TIMESTAMP,
    edited_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE,
    like_count INT DEFAULT 0,

    FOREIGN KEY (post_id) REFERENCES posts (post_id),
    FOREIGN KEY (user_id) REFERENCES users (user_id),
    FOREIGN KEY (parent_comment_id) REFERENCES comments(comment_id)
);

CREATE TABLE community_mods(
    moderator_id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    community_id INT NOT NULL,
    user_id VARCHAR(255) NOT NULL,
    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    UNIQUE (community_id, user_id),
    
    FOREIGN KEY (community_id) REFERENCES communities (community_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE categories(
    category_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    parent_category_id INT, 
    name VARCHAR(255) NOT NULL UNIQUE, 

    FOREIGN KEY (parent_category_id) REFERENCES categories (category_id)
);

CREATE TYPE condition_type AS ENUM ('Used', 'New');

CREATE TABLE listings(
    listing_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL, 
    category_id INT NOT NULL, 
    description TEXT,
    title VARCHAR(255) NOT NULL,
    item_type VARCHAR(255) NOT NULL,
    star_rating INT,
    price DECIMAL(10, 2) NOT NULL, 
    date_posted DATE DEFAULT CURRENT_DATE,
    condition condition_type NOT NULL,
    image_url VARCHAR(255) NOT NULL, 

    FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (category_id) REFERENCES categories (category_id)
);

CREATE TABLE favorites(
    user_id VARCHAR(255) NOT NULL,
    listing_id BIGINT NOT NULL,
    favorited_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    PRIMARY KEY (user_id, listing_id),
    
    FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (listing_id) REFERENCES listings (listing_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE scheduling(
    appointment_id BIGINT GENERATED BY ALWAYS AS IDENTITY PRIMARY KEY,
    listing_id BIGINT NOT NULL,
    seller_id VARCHAR(255) NOT NULL,
    buyer_id VARCHAR(255) NOT NULL,
    location VARCHAR(2048) NOT NULL,
    date DATE NOT NULL, 
    time TIME NOT NULL,
    
    FOREIGN KEY (listing_id) REFERENCES listings (listing_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (seller_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (buyer_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE transactions(
    transaction_id BIGINT GENERATED BY ALWAYS AS IDENTITY PRIMARY KEY,
    listing_id BIGINT NOT NULL,
    seller_id VARCHAR(255) NOT NULL,
    buyer_id VARCHAR(255) NOT NULL,
    transaction_date DATE DEFAULT CURRENT_DATE,
    location VARCHAR(2048) NOT NULL, 

    FOREIGN KEY (listing_id) REFERENCES listings (listing_id),
    FOREIGN KEY (seller_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (buyer_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE messages(
    message_id BIGINT GENERATED BY ALWAYS AS IDENTITY PRIMARY KEY,
    listing_id BIGINT,
    sender_id VARCHAR(255) NOT NULL,
    receiver_id VARCHAR(255) NOT NULL,
    content TEXT NOT NULL, 
    time_stamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_read BOOL DEFAULT FALSE,

    FOREIGN KEY (listing_id) REFERENCES listings (listing_id),
    FOREIGN KEY (sender_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (receiver_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE reviews(
    review_id BIGINT GENERATED BY ALWAYS AS IDENTITY PRIMARY KEY,
    reviewer_id VARCHAR(255) NOT NULL,
    reviewee_id VARCHAR(255) NOT NULL,
    content TEXT, 
    rating INT NOT NULL CHECK (rating <= 5 and rating >= 0),

    FOREIGN KEY (reviewer_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (reviewee_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);
